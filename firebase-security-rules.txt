// ============================================
// PRODUKCYJNE REGUŁY BEZPIECZEŃSTWA FIRESTORE
// ============================================
//
// Jak wdrożyć:
// 1. Wejdź do konsoli Firebase -> Firestore Database -> Rules.
// 2. Skopiuj poniższe reguły i kliknij "Publish".
// 3. JEDNORAZOWY BOOTSTRAP ADMINISTRATORA:
//    Aby pierwszy administrator miał uprawnienia do zapisu chronionych kolekcji,
//    w konsoli Firebase -> Firestore utwórz kolekcję 'admins', a w niej dokument o ID
//    równym Twojemu UID z Firebase Authentication, z polem: { role: 'admin' }.
//    Po zalogowaniu pierwszego admina moduł Opcje automatycznie zsynchronizuje pozostałych adminów.
//
// ============================================

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ============================================
    // FUNKCJE POMOCNICZE
    // ============================================

    function isAuthenticated() {
      return request.auth != null;
    }

    // Weryfikacja uprawnień administratora na podstawie kolekcji admins/{uid}
    function isAdmin() {
      return isAuthenticated() && exists(/databases/$(database)/documents/admins/$(request.auth.uid));
    }

    // Pomocnicza reguła dla aktualizacji grafiku przez zalogowanych pracowników
    function isScheduleCellsUpdate() {
      return request.resource.data.diff(resource.data).affectedKeys()
        .hasOnly(['scheduleCells', 'lastSavedAt', 'lastSavedBy', 'saveToken']);
    }

    // Pomocnicza reguła dla użytkownika w trybie masażu przełączającego wyłącznie własny accessMode
    function isSelfAccessModeUpdate() {
      let idx = request.resource.data.lastSelfUpdatedIndex;
      return request.resource.data.diff(resource.data).affectedKeys().hasOnly(['employees', 'lastSelfUpdatedIndex'])
        && idx is string
        && idx in resource.data.employees
        && idx in request.resource.data.employees
        && request.resource.data.employees.diff(resource.data.employees).affectedKeys().hasOnly([idx])
        && resource.data.employees[idx].uid == request.auth.uid
        && request.resource.data.employees[idx].diff(resource.data.employees[idx]).affectedKeys().hasOnly(['accessMode'])
        && request.resource.data.employees[idx].accessMode in ['massage', 'full'];
    }

    // ============================================
    // REGUŁY DLA DOKUMENTÓW I KOLEKCJI
    // ============================================

    // Główny dokument grafiku
    match /schedules/mainSchedule {
      allow read: if isAuthenticated();
      allow create: if isAdmin();
      allow update: if isAdmin() || (isAuthenticated() && (isScheduleCellsUpdate() || isSelfAccessModeUpdate()));
      allow delete: if isAdmin();
    }

    // Harmonogramy zmian (np. changesSchedule_2025, changesSchedule_2026)
    // WAŻNE: wykluczamy 'mainSchedule', aby reguła z wildcardem nie nadpisała ochrony głównego dokumentu!
    match /schedules/{scheduleId} {
      allow read: if isAuthenticated();
      allow write: if isAuthenticated() && scheduleId != 'mainSchedule';
    }

    // Urlopy
    match /leaves/{document} {
      allow read, write: if isAuthenticated();
    }

    // Planowanie zabiegów
    match /appointments/{document} {
      allow read, write: if isAuthenticated();
    }

    // Stanowiska i pokoje
    match /stations/{document} {
      allow read, write: if isAuthenticated();
    }

    // Statystyki tygodniowe i dzienne
    match /statsWeekly/{document} {
      allow read, write: if isAuthenticated();
    }

    match /statsDaily/{document} {
      allow read, write: if isAuthenticated();
    }

    // Kopia zapasowa - wyłącznie administrator
    match /backup/{document} {
      allow read, write: if isAdmin();
    }

    match /backups/{document} {
      allow read, write: if isAdmin();
    }

    // Kolekcja administratorów - odczyt dla zalogowanych, zapis tylko dla admina
    match /admins/{userId} {
      allow read: if isAuthenticated();
      allow write: if isAdmin();
    }

    // Test połączenia (dla connection monitor)
    match /_connection_test/{document} {
      allow read, write: if isAuthenticated();
    }
  }
}
